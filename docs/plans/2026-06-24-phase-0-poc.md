# ICD-10 Vietnam Explorer — Phase 0 (PoC) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working Proof of Concept with 120-150 representative ICD codes covering all 5 disease groups, demonstrating the full architecture from Build Pipeline → Search → Detail View → Explain Engine.

**Architecture:** Build Pipeline (Node.js scripts) converts raw `icd10_flat.json` into typed, pre-compiled `/public/build/*.json` files. A Vite + React PWA loads this data into IndexedDB (Dexie.js) and serves Search (Orama), Detail View (8 tabs), Hierarchy Browser, Rule Playground, and Explain Engine — all 100% offline.

**Tech Stack:** Node.js (build scripts), Vite + React 18 + TypeScript, Dexie.js (IndexedDB), Orama (search), Tailwind CSS + shadcn/ui, Vitest (tests)

**PoC Dataset (5 groups, ~140 codes):**
- Group 1 (Khám thai): Z30–Z39
- Group 2 (Chấn thương + Nguyên nhân ngoài): S00–S09, V01–V99 (chọn lọc ~30 mã)
- Group 3 (Dagger/Asterisk): A50–A52, I98
- Group 4 (Combination coding): E10–E14, N18
- Group 5 (Ung thư): C00–C10 (chọn lọc ~15 mã)

**Go/No-Go Criteria (must ALL pass before Phase 1):**
- Search latency < 30ms
- Detail View render < 50ms
- Rules accuracy 100% on PoC test set
- 100% provenance traceability
- Explain Engine (no LLM) handles ≥90% PoC queries
- LLM scope restricted to HSBA/multi-disease cases

```typescript
import { readFileSync } from 'fs'
import { resolve } from 'path'
import type { ICDRecord } from '../../src/types/icd'

const POC_CODE_PREFIXES = [
  // Group 1: Khám thai
  'Z30','Z31','Z32','Z33','Z34','Z35','Z36','Z37','Z38','Z39',
  // Group 2: Chấn thương đầu + Nguyên nhân giao thông
  'S00','S01','S02','S06','S09',
  'V01','V02','V03','V10','V19','V20','V23','V40','V60','V89','V99',
  // Group 3: Giang mai (Dagger/Asterisk)
  'A50','A51','A52','I98',
  // Group 4: ĐTĐ + Thận (Combination)
  'E10','E11','E12','E13','E14','N18',
  // Group 5: Ung thư (chọn lọc)
  'C00','C01','C02','C03','C04','C05','C06','C07','C08','C09','C10',
]

function normalizeBool(val: unknown): boolean {
  if (val === null || val === undefined || val === '') return false
  return true
}

export async function extractPocDataset(): Promise<ICDRecord[]> {
  const raw = JSON.parse(
    readFileSync(resolve(__dirname, '../../icd10_flat.json'), 'utf-8')
  )
  
  const records: ICDRecord[] = []
  
  for (const row of raw) {
    const maBenh = row?.benh?.maBenh
    if (!maBenh || maBenh === 'MÃ BỆNH') continue
    
    const matchesPrefix = POC_CODE_PREFIXES.some(prefix => 
      String(maBenh).startsWith(prefix)
    )
    if (!matchesPrefix) continue
    
    records.push({
      maBenh: String(maBenh),
      maBenhKhongDau: String(row.benh.maBenhKhongDau || maBenh),
      tenTiengViet: String(row.benh.tenTiengViet || ''),
      tenTiengAnh: String(row.benh.tenTiengAnh || ''),
      huongDanMaHoaTiengViet: row.benh.huongDanMaHoaTiengViet || null,
      huongDanMaHoaTiengAnh: row.benh.huongDanMaHoaTiengAnh || null,
      chuongStt: String(row.chuong?.stt || ''),
      chuongPhamViMa: String(row.chuong?.phamViMa || ''),
      chuongTenViet: String(row.chuong?.tenTiengViet || ''),
      chuongTenAnh: String(row.chuong?.tenTiengAnh || ''),
      khoiMa: String(row.khoi?.ma || ''),
      khoiTenViet: String(row.khoi?.tenTiengViet || ''),
      khoiTenAnh: String(row.khoi?.tenTiengAnh || ''),
      nhomMa: String(row.nhomBenh3KyTu?.ma || ''),
      nhomTenViet: String(row.nhomBenh3KyTu?.tenTiengViet || ''),
      nhomTenAnh: String(row.nhomBenh3KyTu?.tenTiengAnh || ''),
      dieuKienSuDung: {
        khongDungLaBenhChinh: normalizeBool(row.benh.dieuKienSuDung?.khongDungLaBenhChinh),
        khongKhuyenKhichDungLaBenhChinh: normalizeBool(row.benh.dieuKienSuDung?.khongKhuyenKhichDungLaBenhChinh),
        khongSuDungViCoMaCuTheHon: normalizeBool(row.benh.dieuKienSuDung?.khongSuDungViCoMaCuTheHon),
```

```typescript
// scripts/build/build-rules.ts
import type { ICDRecord, ICDRule, SourceReference } from '../../src/types/icd'

const PROVENANCE_BASE: SourceReference = {
  source: 'icd10_flat',
  confidence: 'exact',
  citationLevel: 'official',
  file: 'icd10_flat.json',
}

export function buildRules(records: ICDRecord[]): ICDRule[] {
  const rules: ICDRule[] = []
  
  for (const rec of records) {
    const d = rec.dieuKienSuDung
    
    if (d.khongDungLaBenhChinh) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'khongDungLaBenhChinh',
        severity: 'error',
        message: `Mã ${rec.maBenh} không được sử dụng làm bệnh chính.`,
        provenance: PROVENANCE_BASE,
      })
    }
    if (d.khongKhuyenKhichDungLaBenhChinh) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'khongKhuyenKhichDungLaBenhChinh',
        severity: 'warning',
        message: `Mã ${rec.maBenh} không khuyến khích dùng làm bệnh chính.`,
        provenance: PROVENANCE_BASE,
      })
    }
    if (d.khongSuDungViCoMaCuTheHon) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'khongSuDungViCoMaCuTheHon',
        severity: 'warning',
        message: `Mã ${rec.maBenh} không nên dùng vì có mã cụ thể hơn.`,
        provenance: PROVENANCE_BASE,
      })
    }
    if (d.chiSuDungMaHoaNguyenNhanTuVong) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'chiSuDungMaHoaNguyenNhanTuVong',
        severity: 'error',
        message: `Mã ${rec.maBenh} chỉ dùng để mã hóa nguyên nhân tử vong.`,
        provenance: PROVENANCE_BASE,
      })
    }
    if (d.chiCoONuGioi) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'chiCoONuGioi',
        severity: 'warning',
        message: `Mã ${rec.maBenh} chỉ áp dụng cho người bệnh nữ.`,
        provenance: PROVENANCE_BASE,
      })
    }
    if (d.chiCoONamGioi) {
      rules.push({
        code: rec.maBenh,
        ruleType: 'chiCoONamGioi',
        severity: 'warning',
        message: `Mã ${rec.maBenh} chỉ áp dụng cho người bệnh nam.`,
        provenance: PROVENANCE_BASE,
      })
    }
  }
  
  return rules
}
```

```typescript
// scripts/build/build-hierarchy.ts
import type { ICDRecord, ICDHierarchy } from '../../src/types/icd'

export function buildHierarchy(records: ICDRecord[]): ICDHierarchy[] {
  const hierarchyMap = new Map<string, ICDHierarchy>()
  
  for (const rec of records) {
    const code = rec.maBenh
    const nhom = rec.nhomMa  // 3-char group e.g. Z34
    
    if (!hierarchyMap.has(code)) {
      hierarchyMap.set(code, {
        code,
        parentCode: nhom !== code ? nhom : rec.khoiMa || null,
        childCodes: [],
        siblingCodes: [],
        level: code.length === 3 ? 'category' : 'subcategory',
        chapterRoman: rec.chuongStt,
        blockRange: rec.khoiMa,
      })
    }
    
    // Register 3-char group node if not exists
    if (nhom && nhom !== code && !hierarchyMap.has(nhom)) {
      hierarchyMap.set(nhom, {
        code: nhom,
        parentCode: rec.khoiMa || null,
        childCodes: [],
        siblingCodes: [],
        level: 'category',
        chapterRoman: rec.chuongStt,
        blockRange: rec.khoiMa,
      })
    }
    
    // Link child → parent
    if (nhom && nhom !== code) {
      const parentNode = hierarchyMap.get(nhom)!
      if (!parentNode.childCodes.includes(code)) {
        parentNode.childCodes.push(code)
      }
    }
  }
  
  // Populate siblings
  for (const node of hierarchyMap.values()) {
    if (node.parentCode) {
      const parent = hierarchyMap.get(node.parentCode)
      if (parent) {
        node.siblingCodes = parent.childCodes.filter(c => c !== node.code)
      }
    }
  }
  
  return Array.from(hierarchyMap.values())
}
```

```typescript
// scripts/build/index.ts
import { writeFileSync, mkdirSync } from 'fs'
import { resolve } from 'path'
import { extractPocDataset } from './extract-poc-dataset'
import { buildHierarchy } from './build-hierarchy'
import { buildRules } from './build-rules'
import { buildCodingRelations, buildInfoRelations } from './build-relations'
import { buildConcepts } from './build-concepts'

const OUT_DIR = resolve(__dirname, '../../public/build')

function writeJson(filename: string, data: unknown, recordCount?: number) {
  const header = {
    schemaVersion: '1.0.0',
    generatedAt: new Date().toISOString(),
    sourceFiles: ['icd10_flat.json'],
    recordCount: recordCount ?? (Array.isArray(data) ? data.length : 1),
  }
  const payload = Array.isArray(data) ? data : data
  writeFileSync(
    resolve(OUT_DIR, filename),
    JSON.stringify({ meta: header, data: payload }, null, 2),
    'utf-8'
  )
  console.log(`✅ ${filename} — ${header.recordCount} records`)
}

async function main() {
  mkdirSync(OUT_DIR, { recursive: true })
  console.log('🔨 Building PoC dataset...')

  const records = await extractPocDataset()
  console.log(`📦 Extracted ${records.length} PoC records`)

  writeJson('search_index.json', records, records.length)
  writeJson('hierarchy.json', buildHierarchy(records))
  writeJson('rules.json', buildRules(records))
  writeJson('coding_relations.json', buildCodingRelations())
  writeJson('info_relations.json', buildInfoRelations())
  writeJson('concepts.json', buildConcepts())

  console.log('✅ Build complete → public/build/')
}

main().catch(console.error)
```

```typescript
import { create, insert, search as oramaSearch } from '@orama/orama'
import type { ICDRecord } from '../types/icd'

let db: Awaited<ReturnType<typeof create>> | null = null

export async function buildSearchIndex(records: ICDRecord[]) {
  db = await create({
    schema: {
      maBenh: 'string',
      maBenhKhongDau: 'string',
      tenTiengViet: 'string',
      tenTiengAnh: 'string',
      huongDanMaHoaTiengViet: 'string',
    } as const,
  })
  
  for (const rec of records) {
    await insert(db, {
      maBenh: rec.maBenh,
      maBenhKhongDau: rec.maBenhKhongDau,
      tenTiengViet: rec.tenTiengViet,
      tenTiengAnh: rec.tenTiengAnh,
      huongDanMaHoaTiengViet: rec.huongDanMaHoaTiengViet ?? '',
    })
  }
}

export async function search(query: string, limit = 20): Promise<ICDRecord[]> {
  if (!db) throw new Error('Search index not initialized')
  
  const results = await oramaSearch(db, {
    term: query,
    limit,
    tolerance: 1, // fuzzy
  })
  
  return results.hits.map(h => h.document as unknown as ICDRecord)
}
```

```typescript
import type { ICDRecord, ICDHierarchy, ICDRule, CodingRelation, ClinicalConcept, ICDVersion } from '../types/icd'

export class ICDDatabase extends Dexie {
  records!: Table<ICDRecord>
  hierarchy!: Table<ICDHierarchy>
  rules!: Table<ICDRule & { id?: number }>
  codingRelations!: Table<CodingRelation & { id?: number }>
  concepts!: Table<ClinicalConcept>

  constructor() {
    super('icd10-vietnam')
    this.version(1).stores({
      records: 'maBenh, maBenhKhongDau, *tenTiengViet',
      hierarchy: 'code, parentCode',
      rules: '++id, code, ruleType',
      codingRelations: '++id, source, target, relationType',
      concepts: 'id, concept',
    })
  }
}

export const icdDb = new ICDDatabase()

export async function seedDatabase() {
  const count = await icdDb.records.count()
  if (count > 0) return // already seeded
  
  const [searchIndex, hierarchyData, rulesData, relationsData, conceptsData] = await Promise.all([
    fetch('/build/search_index.json').then(r => r.json()),
    fetch('/build/hierarchy.json').then(r => r.json()),
    fetch('/build/rules.json').then(r => r.json()),
    fetch('/build/coding_relations.json').then(r => r.json()),
    fetch('/build/concepts.json').then(r => r.json()),
  ])
  
  await icdDb.records.bulkPut(searchIndex.data)
  await icdDb.hierarchy.bulkPut(hierarchyData.data)
  await icdDb.rules.bulkPut(rulesData.data)
  await icdDb.codingRelations.bulkPut(relationsData.data)
  await icdDb.concepts.bulkPut(conceptsData.data)
}
```

```typescript
// src/lib/explain.ts
import type { ExplainDecision, ICDRule, ClinicalConcept, SourceReference } from '../types/icd'

export async function explainCode(
  code: string,
  rules: ICDRule[],
  concepts: ClinicalConcept[]
): Promise<ExplainDecision> {
  const applicableRules = rules.filter(r => r.code === code)
  
  const provenance: SourceReference[] = applicableRules.length > 0
    ? applicableRules.map(r => r.provenance)
    : [{
        source: 'icd10_flat',
        confidence: 'exact',
        citationLevel: 'official',
        file: 'icd10_flat.json',
      }]
  
  const reasons = applicableRules.length > 0
    ? applicableRules.map(r => r.message)
    : [`Mã ${code} được tìm thấy trong cơ sở dữ liệu ICD-10 Việt Nam.`]
  
  return {
    query: code,
    selectedCodes: [{
      code,
      confidence: 1.0,
      reasons,
      provenance,
    }],
    rejectedCodes: [],
    rulesApplied: applicableRules.map(r => r.ruleType),
    mode: 'template',
  }
}

export async function explainConcept(
  query: string,
  rules: ICDRule[],
  concepts: ClinicalConcept[]
): Promise<ExplainDecision> {
  // Find matching concept by alias
  const queryLower = query.toLowerCase().trim()
  let matchedConcept: ClinicalConcept | undefined
  let matchScore = 0
  
  for (const concept of concepts) {
    for (const alias of concept.aliases) {
      if (alias.status !== 'active') continue
      if (queryLower.includes(alias.text.toLowerCase()) || alias.text.toLowerCase().includes(queryLower)) {
        const score = alias.weight
        if (score > matchScore) {
          matchScore = score
          matchedConcept = concept
        }
      }
    }
  }
  
  if (!matchedConcept) {
    return {
      query,
      selectedCodes: [],
      rejectedCodes: [],
      rulesApplied: [],
      mode: 'template',
    }
  }
  
  // Sort candidates by score
  const sorted = [...matchedConcept.candidateCodes].sort((a, b) => b.score - a.score)
  const topCandidate = sorted[0]
  const rejected = sorted.slice(1)
  
  const codeRules = rules.filter(r => r.code === topCandidate.code)
  
  return {
    query,
    selectedCodes: [{
      code: topCandidate.code,
      confidence: topCandidate.score,
      reasons: [
        `Khớp với khái niệm: "${matchedConcept.concept}"`,
        ...codeRules.map(r => r.message),
        ...(topCandidate.conditions ? [`Điều kiện: ${topCandidate.conditions.join(', ')}`] : []),
      ],
      provenance: [{
        source: 'concept_dictionary',
        confidence: 'derived',
        citationLevel: 'inferred',
        file: 'concepts.json',
      }],
    }],
    rejectedCodes: rejected.map(r => ({
      code: r.code,
      confidence: r.score,
      reason: r.conditions
        ? `Yêu cầu điều kiện: ${r.conditions.join(', ')}`
        : r.exclusions
          ? `Bị loại trừ nếu có: ${r.exclusions.join(', ')}`
          : 'Điểm phù hợp thấp hơn mã được chọn',
    })),
    rulesApplied: codeRules.map(r => r.ruleType),
    mode: 'template',
  }
}
```
